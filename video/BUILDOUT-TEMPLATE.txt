STEP 6: VIDEO TEMPLATES (5 universal templates)
==============================================================================

These 5 templates work for ANY B2B/B2C SaaS:

TEMPLATE 1: "Did You Know" (Fact/Stat)
────────────────────────────────────────
Purpose: Educate + surprise → drive curiosity
Structure: Hook (2s) → Niche icon (1s) → Stat counter (4s) → Explanation (5s) → CTA (4s)
Works for: ANY niche with data/stats
Example hooks:
  - "[STAT]% of [AUDIENCE] still do [THING] manually"
  - "The average [ROLE] wastes [X] hours/week on [TASK]"
  - "[AUDIENCE] who use [FEATURE] save [AMOUNT]"

TEMPLATE 2: "Price/Comparison Check"
────────────────────────────────────────
Purpose: Show savings → justify the app
Structure: Hook (2s) → Side-by-side bars (6s) → Savings callout (4s) → CTA (4s)
Works for: Any app that saves money (pricing, sourcing, comparison)
Example hooks:
  - "Are you overpaying for [THING]?"
  - "[OPTION A] vs [OPTION B] — which saves more?"

TEMPLATE 3: "Before / After"
────────────────────────────────────────
Purpose: Visual proof → desire
Structure: Before photo (3s) → Swipe (1s) → After photo (3s) → Overlay (2s) → CTA (3s)
Works for: ANY app with visual transformation (design, construction, fitness, cooking)
Example hooks:
  - "What [X] hours of work looks like"
  - "From [BAD_STATE] to [GOOD_STATE]"

TEMPLATE 4: "AI/App Says" (Tip)
────────────────────────────────────────
Purpose: Position app as smart advisor → trust
Structure: Logo (2s) → Tip typewriter (5s) → CTA (3s)
Works for: ANY app with AI or smart features
Example hooks:
  - "This one trick [BENEFIT]"
  - "[APP_NAME] just told me something I didn't know"

TEMPLATE 5: "Deadline/Alert"
────────────────────────────────────────
Purpose: Urgency → fear of missing out → download
Structure: Context (2s) → Countdown (4s) → Consequence (4s) → Solution (4s) → CTA (4s)
Works for: Compliance, deadlines, seasonal, limited offers
Example hooks:
  - "This [THING] expires next month. Are you ready?"
  - "Only [X] days left to [ACTION]"

TEMPLATE ANATOMY (copy-paste starter)

// video/src/compositions/MyTemplate.tsx

import React from 'react';
import {
  AbsoluteFill,
  interpolate,
  useCurrentFrame,
  useVideoConfig,
  spring,
  Sequence,
} from 'remotion';
import { Colors } from '../data/theme';
import { CTABanner } from '../components/Branding';

export interface MyTemplateProps {
  hookText: string;           // Grab attention in first 2 seconds
  mainContent: string;        // The value — what the viewer learns
  ctaText: string;            // Call to action
  language: string;           // For multilingual support
}

export const MyTemplate: React.FC<MyTemplateProps> = ({
  hookText, mainContent, ctaText,
}) => {
  const frame = useCurrentFrame();
  const { fps } = useVideoConfig();

  // Animation timeline (21 seconds = 630 frames at 30fps)
  // Phase 1: Hook (frames 0-60, 2 seconds)
  const hookOpacity = interpolate(
    frame, [0, 10, fps * 2 - 10, fps * 2],
    [0, 1, 1, 0], { extrapolateRight: 'clamp' }
  );

  // Phase 2: Main content (frames 60-420, 12 seconds)
  const contentOpacity = interpolate(
    frame, [fps * 2, fps * 3],
    [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Phase 3: CTA (frames 420-570, 5 seconds)
  const ctaOpacity = interpolate(
    frame, [fps * 14, fps * 15],
    [0, 1], { extrapolateLeft: 'clamp', extrapolateRight: 'clamp' }
  );

  // Phase 4: Branding (frames 570-630, 2 seconds)
  // (handled by VascoBranding component)

  return (
    <AbsoluteFill style={{ backgroundColor: Colors.background }}>
      {/* HOOK — first 2 seconds, biggest text, grab attention */}
      <Sequence from={0} durationInFrames={fps * 3}>
        <AbsoluteFill style={{
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          padding: 60, opacity: hookOpacity,
        }}>
          <span style={{
            fontSize: 48, fontWeight: 800, color: Colors.textPrimary,
            fontFamily: 'Inter, sans-serif', textAlign: 'center',
            lineHeight: 1.2, letterSpacing: -1,
          }}>
            {hookText}
          </span>
        </AbsoluteFill>
      </Sequence>

      {/* MAIN CONTENT — the value the viewer gets */}
      <Sequence from={fps * 2} durationInFrames={fps * 12}>
        <AbsoluteFill style={{
          display: 'flex', alignItems: 'center', justifyContent: 'center',
          padding: 60, opacity: contentOpacity,
        }}>
          <span style={{
            fontSize: 30, fontWeight: 500, color: Colors.textPrimary,
            fontFamily: 'Inter, sans-serif', textAlign: 'center',
            lineHeight: 1.5,
          }}>
            {mainContent}
          </span>
        </AbsoluteFill>
      </Sequence>

      {/* CTA — what the viewer should do */}
      <Sequence from={fps * 14} durationInFrames={fps * 5}>
        <div style={{ opacity: ctaOpacity }}>
          <CTABanner text={ctaText} />
        </div>
      </Sequence>
    </AbsoluteFill>
  );
};

==============================================================================
ANIMATION CHEAT SHEET
==============================================================================

FADE IN:
  interpolate(frame, [startFrame, startFrame + 10], [0, 1], { extrapolateRight: 'clamp' })

FADE OUT:
  interpolate(frame, [endFrame - 10, endFrame], [1, 0], { extrapolateLeft: 'clamp' })

SCALE BOUNCE:
  spring({ frame: frame - startFrame, fps, config: { damping: 12 } })

SLIDE UP:
  interpolate(frame, [startFrame, startFrame + 15], [100, 0], { extrapolateRight: 'clamp' })

COUNTER (animate number):
  Math.floor(interpolate(frame, [start, end], [0, targetNumber], { extrapolateRight: 'clamp' }))

TYPEWRITER:
  text.slice(0, Math.floor(interpolate(frame, [start, end], [0, text.length])))

PULSE:
  Math.sin(frame * 0.15) * 0.1 + 1  // gentle 1.0 ± 0.1
